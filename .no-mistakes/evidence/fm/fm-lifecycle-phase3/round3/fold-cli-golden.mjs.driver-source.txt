import assert from 'node:assert/strict';import {mkdtempSync,mkdirSync,writeFileSync,readFileSync,existsSync,rmSync} from 'node:fs';import {join} from 'node:path';import {spawnSync} from 'node:child_process';
const root=process.cwd(),evidence=process.env.FM_TEST_EVIDENCE,work=join(root,'.phase3-live/fold-cli');mkdirSync(work,{recursive:true});mkdirSync(join(work,'user-home'),{recursive:true});const home=mkdtempSync(join(process.env.TMPDIR||'/tmp','fm-lab.'));const env={PATH:'/usr/bin:/bin:/usr/sbin:/sbin',HOME:join(work,'user-home'),TMPDIR:process.env.TMPDIR||'/tmp',LANG:'C',LC_ALL:'C',FM_HOME:home,FM_SUPERVISION_ACTOR:'main',FM_WEDGE_ALARM_EXEC:'discard'};
const run=(cmd,args)=>{const r=spawnSync(cmd,args,{cwd:root,env,encoding:'utf8',timeout:90000});if(r.status!==0)throw Error(`${cmd}: ${r.error||''}${r.stderr}${r.stdout}`);return r.stdout};const rows=text=>text.split('\n').filter(l=>/^task \[key=/.test(l)).sort();const observations=[];let tmuxStarted=false;
try{run('bin/fm-lab-home.sh',['create',home]);mkdirSync(join(home,'tmux'));env.TMUX_TMPDIR=join(home,'tmux');const f=join(home,'state/task.status'),c=join(home,'state/.task.open-decisions-cursor');
 for(const kind of ['ship','scout','secondmate']){rmSync(c,{force:true});writeFileSync(join(home,'state/task.meta'),`kind=${kind}\n`);writeFileSync(f,readFileSync(join(evidence,`fold-live/logs/golden-${kind}-prefix.status`)));run('bin/fm-wake-drain.sh',[]);writeFileSync(f,readFileSync(join(evidence,`fold-live/logs/golden-${kind}-tail.bytes`)),{flag:'a'});const seeded=run('bin/fm-wake-drain.sh',[]);writeFileSync(join(evidence,`fold-cli-${kind}-seeded.txt`),seeded);rmSync(c);const unseeded=run('bin/fm-wake-drain.sh',[]);writeFileSync(join(evidence,`fold-cli-${kind}-unseeded.txt`),unseeded);assert.deepEqual(rows(seeded),rows(unseeded));const keys=rows(seeded).map(l=>l.match(/\[key=([^\]]+)\]/)[1]).sort();assert.deepEqual(keys,kind==='secondmate'?['beta','delta','gamma','pending-reply-abcd']:['delta']);observations.push({scenario:`actual ${kind} seeded and unseeded drain equivalence`,keys,rows:rows(seeded)})}
 const before=readFileSync(c,'utf8');const snapshotScript=join(work,'snapshot.sh');writeFileSync(snapshotScript,`#!/bin/bash\nbin/fm-fleet-snapshot.sh --home-input > '${join(evidence,'fold-fleet-home-input.json')}' 2> '${join(evidence,'fold-fleet-home-input.stderr')}'\nprintf '%s\\n' "$?" > '${join(work,'snapshot.rc')}'\n`);const tmux=(...a)=>run('/opt/homebrew/bin/tmux',['-L','fm-lab',...a]);tmux('new-session','-d','-s','snapshot','-x','120','-y','40','-c',root,'-e',`FM_HOME=${home}`,`/bin/bash '${snapshotScript}'`);tmuxStarted=true;for(let i=0;i<900&&!existsSync(join(work,'snapshot.rc'));i++)await new Promise(r=>setTimeout(r,100));assert.equal(readFileSync(join(work,'snapshot.rc'),'utf8').trim(),'0',readFileSync(join(evidence,'fold-fleet-home-input.stderr'),'utf8'));const snapshot=JSON.parse(readFileSync(join(evidence,'fold-fleet-home-input.json'),'utf8'));assert.equal(snapshot.schema,'fm-fleet-home-input.v1');const task=snapshot.tasks.find(t=>t.id==='task');assert.ok(task,JSON.stringify(snapshot));const keys=task.hints.open_decisions.map(d=>d.key).sort();assert.deepEqual(keys,['beta','delta','gamma','pending-reply-abcd']);assert.equal(readFileSync(c,'utf8'),before);observations.push({scenario:'actual fleet snapshot carry preserves secondmate decisions and live checkpoint',keys,checkpointUnchanged:true});console.log(JSON.stringify(observations,null,2));
}finally{writeFileSync(join(evidence,'fold-cli-golden-observations.json'),JSON.stringify(observations,null,2)+'\n');if(tmuxStarted)spawnSync('/opt/homebrew/bin/tmux',['-L','fm-lab','kill-server'],{cwd:root,env,encoding:'utf8'});rmSync(home,{recursive:true,force:true})}
