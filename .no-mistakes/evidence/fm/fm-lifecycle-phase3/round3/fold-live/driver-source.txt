def save(name,data):
    p=E/name; p.parent.mkdir(parents=True,exist_ok=True)
    p.write_bytes(data if isinstance(data,bytes) else data.encode())
    return str(p)

def record(name,ok,details):
    results.append({'scenario':name,'live':True,'result':'pass' if ok else 'FAIL','details':details})

def home(name):
    h=F/name; (h/'state').mkdir(parents=True); (h/'data').mkdir(); (h/'config').write_text('')
    return h

def env_for(h,probe):
    return dict(base,FM_HOME=str(h),FM_ROOT_OVERRIDE=str(h),FM_STATE_OVERRIDE=str(h/'state'),FM_DATA_OVERRIDE=str(h/'data'),FM_CONFIG_OVERRIDE=str(h/'config'),FM_OPEN_DECISIONS_READ_PROBE=str(probe))

def run(name,args,env,timeout=180):
    commands.append({'name':name,'argv':list(map(str,args)),'cwd':str(ROOT),'env':env})
    started=time.perf_counter(); p=subprocess.run(args,cwd=ROOT,env=env,capture_output=True,timeout=timeout)
    elapsed=time.perf_counter()-started
    save(name+'.stdout',p.stdout); save(name+'.stderr',p.stderr)
    save(name+'.run.json',json.dumps({'exit_code':p.returncode,'wall_seconds':elapsed},indent=2))
    return p,elapsed

def state_copy(label,h):

    if not probe.exists(): return []
